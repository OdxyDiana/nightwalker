<script type="text/javascript">
    // Avatar y nombre del usuario conectado (mismo origen de datos que switch-account.js)
    (function () {
        var user = window._userdata;
        var box = document.getElementById('header-user');
        if (!user || Number(user.session_logged_in) !== 1 || !box) return;

        var avatar = user.avatar_link || '';
        if (!avatar && user.avatar) {
            var match = String(user.avatar).match(/src=["']([^"']+)["']/);
            if (match) avatar = match[1];
        }
        var name = String(user.username || '');
        if (user.user_id) box.href = '/u' + user.user_id;
        box.setAttribute('data-title', name);

        var frame = document.createElement('div');
        frame.className = 'shrink-0 w-[40px] h-[40px] sm:w-[50px] sm:h-[50px] overflow-hidden rounded-sm surface-2 p-[5px] object-cover object-center border border-color-3 box-content';
        if (avatar) {
            var img = document.createElement('img');
            img.src = avatar;
            img.alt = name;
            img.className = 'w-[40px] h-[40px] sm:w-[50px] sm:h-[50px] border-color-3 border object-cover object-center rounded-sm';
            frame.appendChild(img);
        }
        var label = document.createElement('span');
        label.className = 'hidden sm:block uppercase font-bold ml-[10px] truncate';
        label.textContent = name;
        box.appendChild(frame);
        box.appendChild(label);
    })();
</script>
